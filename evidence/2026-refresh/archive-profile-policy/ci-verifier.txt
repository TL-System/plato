import json
import re
from collections import Counter
from pathlib import Path
from xml.etree import ElementTree

expected_case_names = [
    "test_retained_width_round[anycostfl]",
    "test_retained_width_round[fedrolex]",
    "test_retained_width_round[heterofl]",
    "test_retained_activated_budget[anycostfl]",
    "test_retained_activated_budget[fedrolex]",
    "test_retained_activated_budget[heterofl]",
    "test_sysheterofl_subnet_round",
    "test_dlg_model_update[lenet]",
    "test_dlg_model_update[resnet_18]",
    "test_local_retired_selection[anycostfl]",
    "test_local_retired_selection[fedrolex]",
    "test_local_retired_selection[heterofl]",
    "test_local_retired_selection[dlg]",
    "test_retained_entrypoint_import[anycostfl-root]",
    "test_retained_entrypoint_import[anycostfl-example]",
    "test_retained_entrypoint_import[fedrolex-root]",
    "test_retained_entrypoint_import[fedrolex-example]",
    "test_retained_entrypoint_import[heterofl-root]",
    "test_retained_entrypoint_import[heterofl-example]",
]


def verify_model_search(
    collection: Path, junit: Path, runlog: Path, receipt: Path
) -> int:
    module = "tests/integration/test_retained_model_search.py"
    expected = Counter(f"{module}::{name}" for name in expected_case_names)
    issues = []
    collected = []
    names = []
    totals = {}
    outcomes = Counter()
    try:
        collected = re.findall(r"^(\S+\.py::\S+)$", collection.read_text(), re.M)
        if Counter(collected) != expected:
            issues.append("Collection must contain every expected node exactly once.")
        root = ElementTree.parse(junit).getroot()
        cases = list(root.iter("testcase"))
        names = [case.get("name", "") for case in cases]
        if Counter(names) != Counter(expected_case_names):
            issues.append("JUnit must contain every expected case exactly once.")
        if any(
            case.get("classname") != "tests.integration.test_retained_model_search"
            for case in cases
        ):
            issues.append("JUnit cases must belong to the dedicated module.")
        if any(
            case.find(tag) is not None
            for case in cases
            for tag in ("failure", "error", "skipped")
        ):
            issues.append(
                "Every JUnit case must pass without failure, error, skip or xfail."
            )
        totals = {
            key: sum(int(suite.attrib[key]) for suite in root.iter("testsuite"))
            for key in ("tests", "failures", "errors", "skipped")
        }
        if totals != {"tests": 19, "failures": 0, "errors": 0, "skipped": 0}:
            issues.append("JUnit suite totals must report exactly 19 passing cases.")
        for count, outcome in re.findall(
            r"\b(\d+) (passed|failed|skipped|xfailed|xpassed|errors?)\b",
            runlog.read_text(),
        ):
            outcomes[outcome] += int(count)
        if outcomes != Counter({"passed": 19}):
            issues.append("Execution must report 19 passes and no other outcome.")
    except (OSError, ValueError, KeyError, ElementTree.ParseError) as error:
        issues.append(f"Missing or invalid qualification artifact: {error}")
    record = {
        "schema_version": 1,
        "accepted": not issues,
        "expected_case_names": expected_case_names,
        "collected_nodes": collected,
        "junit_case_names": names,
        "suite_totals": totals,
        "execution_outcomes": dict(outcomes),
        "issues": issues,
    }
    receipt.write_text(json.dumps(record, indent=2) + "\n")
    return int(bool(issues))


raise SystemExit(
    verify_model_search(
        Path("ci-artifacts/model-search-collection.log"),
        Path("ci-artifacts/model-search.xml"),
        Path("ci-artifacts/model-search.log"),
        Path("ci-artifacts/model-search-acceptance.json"),
    )
)
